<script>
  import { onMount } from 'svelte';

  // Typesetting HUD State
  let scale = $state(1);
  let colWidth = $state(68);
  let lineHt = $state(1.75);

  // Tree Canvas Parameters (inspired by Ref 3)
  let treeScale = $state(1.0);
  let varianceAngle = $state(28);
  let leafRotation = $state(45);
  let activeNode = $state({ id: 349, title: "Spatial Memory & Externalized Databanks", likes: 29, comments: 18, date: "2026-07-16" });

  let canvasEl;

  function updateVars() {
    if (typeof document !== 'undefined') {
      document.documentElement.style.setProperty('--essay-font-scale', scale);
      document.documentElement.style.setProperty('--essay-col-width', `${colWidth}ch`);
      document.documentElement.style.setProperty('--essay-line-height', lineHt);
    }
  }

  // Draw Generative Argument Tree on Canvas
  onMount(() => {
    drawTree();
  });

  function drawTree() {
    if (!canvasEl) return;
    const ctx = canvasEl.getContext('2d');
    ctx.clearRect(0, 0, canvasEl.width, canvasEl.height);
    
    // Draw background grid lines
    ctx.strokeStyle = '#12203815';
    ctx.lineWidth = 1;
    for (let i = 0; i < canvasEl.width; i += 30) {
      ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i, canvasEl.height); ctx.stroke();
    }
    for (let j = 0; j < canvasEl.height; j += 30) {
      ctx.beginPath(); ctx.moveTo(0, j); ctx.lineTo(canvasEl.width, j); ctx.stroke();
    }

    // Recursive tree branch drawer
    function branch(x, y, len, angle, depth) {
      if (depth === 0) {
        ctx.fillStyle = '#122038';
        ctx.beginPath();
        ctx.arc(x, y, 3 * treeScale, 0, Math.PI * 2);
        ctx.fill();
        return;
      }
      const endX = x + len * Math.cos(angle);
      const endY = y + len * Math.sin(angle);

      ctx.strokeStyle = '#122038';
      ctx.lineWidth = Math.max(1, depth * 1.5);
      ctx.beginPath();
      ctx.moveTo(x, y);
      ctx.lineTo(endX, endY);
      ctx.stroke();

      const angleSpread = (varianceAngle * Math.PI) / 180;
      branch(endX, endY, len * 0.75, angle - angleSpread, depth - 1);
      branch(endX, endY, len * 0.75, angle + angleSpread, depth - 1);
    }

    // Root start at bottom right
    branch(canvasEl.width - 80, canvasEl.height - 40, 70, -Math.PI / 2, 5);
  }

  $effect(() => {
    // Re-render tree when sliders change
    if (canvasEl) {
      scale, varianceAngle, leafRotation;
      drawTree();
    }
  });
</script>
